enable

int g0/1

no shut
exit

do show ip int brief

end


SWITCH S1
---------

# Management VLAN
enable

# Generate RSA Key
crypto key generate rsa
# Choose 1024 bits

transport input ssh
exit

do show running-config
do wr
end
